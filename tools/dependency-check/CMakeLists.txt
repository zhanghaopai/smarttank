cmake_minimum_required(VERSION 3.29)
project(SmartTankDependencyCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
find_package(OpenCASCADE REQUIRED)
add_executable(dependency-check main.cpp)
target_include_directories(dependency-check PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(dependency-check PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(dependency-check PRIVATE Qt6::Widgets TKernel TKMath TKBRep TKPrim)
enable_testing()
add_test(NAME dependency-check COMMAND dependency-check)
set_tests_properties(dependency-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
